Start in 30 seconds

Use the OpenAI SDK style and swap only the base URL plus the model name.

from openai import OpenAI

client = OpenAI(
    base_url="https://crazyrouter.com/v1",
    api_key="your-crazyrouter-key"
)

response = client.chat.completions.create(
    model="deepseek-r1",
    messages=[{"role": "user", "content": "Solve this planning problem step by step and explain the tradeoffs between the options."}]
)

print(response.choices[0].message.content)
curl https://crazyrouter.com/v1/chat/completions   -H "Authorization: Bearer your-crazyrouter-key"   -H "Content-Type: application/json"   -d '{
    "model": "deepseek-r1",
    "messages": [{"role": "user", "content": "Solve this planning problem step by step and explain the tradeoffs between the options."}]
  }'
import OpenAI from "openai";

const client = new OpenAI({
  baseURL: "https://crazyrouter.com/v1",
  apiKey: "your-crazyrouter-key",
});

const response = await client.chat.completions.create({
  model: "deepseek-r1",
  messages: [{ role: "user", content: "Solve this planning problem step by step and explain the tradeoffs between the options." }],
});

console.log(response.choices[0].message.content);
